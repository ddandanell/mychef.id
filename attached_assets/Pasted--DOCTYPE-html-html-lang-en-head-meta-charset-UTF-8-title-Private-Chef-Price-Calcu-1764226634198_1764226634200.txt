    function formatIDR(value) {
      return "IDR " + value.toLocaleString("id-ID");
    }

    function calculateChefPricing({ guests, mealsPerDay, days, serviceType, cuisineType }) {
      const chefRate = 800000;
      const helperRate = 400000;

      const cuisinePercentage = {
        indonesian: 0.15,
        greek: 0.30,
        italian: 0.35,
        american: 0.40
      };

      // Hours per day based on meals
      let hoursPerDay;
      if (mealsPerDay === 1) hoursPerDay = 4;
      else if (mealsPerDay === 2) hoursPerDay = 6;
      else hoursPerDay = 8;

      // Monthly = always 8h/day
      const effectiveHoursPerDay = serviceType === "month" ? 8 : hoursPerDay;

      // Chef
      const chefHoursTotal = effectiveHoursPerDay * days;
      const chefLabor = chefHoursTotal * chefRate;

      // Helper (>10 guests, 50% of chef hours)
      const helperHoursTotal = guests > 10 ? effectiveHoursPerDay * 0.5 * days : 0;
      const helperLabor = helperHoursTotal * helperRate;

      // Labor total
      const laborTotal = chefLabor + helperLabor;

      // Discount
      let discountRate = 0;
      if (serviceType === "week") discountRate = 0.2;
      else if (serviceType === "month") discountRate = 0.4;

      const laborAfterDiscount = laborTotal * (1 - discountRate);

      // Shopping suggestion (food cost estimate)
      const shoppingPercentage =
        cuisinePercentage[cuisineType] ?? cuisinePercentage.indonesian;
      const suggestedShopping = laborAfterDiscount * shoppingPercentage;

      // Final outputs
      const finalServicePrice = laborAfterDiscount;
      const totalEstimatedCost = finalServicePrice + suggestedShopping;

      return {
        chefLabor,
        helperLabor,
        laborTotal,
        discountRate,
        finalServicePrice,
        suggestedShopping,
        totalEstimatedCost,
        effectiveHoursPerDay,
        chefHoursTotal,
        helperHoursTotal
      };
    }

    function handleCalculate() {
      const guests = parseInt(document.getElementById("guests").value || "0", 10);
      const days = parseInt(document.getElementById("days").value || "0", 10);
      const cuisineType = document.getElementById("cuisineType").value;

      const mealsPerDay = parseInt(
        document.querySelector('input[name="mealsPerDay"]:checked').value,
        10
      );
      const serviceType = document.querySelector('input[name="serviceType"]:checked').value;

      if (!guests || !days) {
        alert("Please enter valid number of guests and days.");
        return;
      }

      const result = calculateChefPricing({
        guests,
        mealsPerDay,
        days,
        serviceType,
        cuisineType
      });

      const resultsDiv = document.getElementById("results");
      resultsDiv.style.display = "block";

      document.getElementById("hoursInfo").textContent =
        `Effective working hours per day: ${result.effectiveHoursPerDay} hours`;

      document.getElementById("chefLabor").textContent =
        `Chef labor: ${formatIDR(result.chefLabor)} (total for ${result.chefHoursTotal} hours)`;

      document.getElementById("helperLabor").textContent =
        result.helperLabor > 0
          ? `Helper labor: ${formatIDR(result.helperLabor)} (total for ${result.helperHoursTotal} hours)`
          : "Helper labor: none (10 guests or fewer)";

      document.getElementById("laborTotal").textContent =
        `Labor total before discount: ${formatIDR(result.laborTotal)}`;

      const discountPercent = result.discountRate * 100;
      document.getElementById("discountInfo").textContent =
        `Discount on labor: ${discountPercent}%`;

      document.getElementById("finalServicePrice").textContent =
        `Final service price (labor after discount): ${formatIDR(result.finalServicePrice)}`;

      document.getElementById("shoppingEstimate").textContent =
        `Estimated food budget (groceries): ${formatIDR(result.suggestedShopping)} (paid according to actual receipt)`;

      document.getElementById("totalEstimate").textContent =
        `Total estimated cost (labor + estimated food): ${formatIDR(result.totalEstimatedCost)}`;
    }
  </script>
