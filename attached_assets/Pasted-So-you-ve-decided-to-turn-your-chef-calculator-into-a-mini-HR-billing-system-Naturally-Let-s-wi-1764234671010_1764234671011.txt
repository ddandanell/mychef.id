4. Full updated calculator function

Here’s the complete JS-style function with your new hourly thresholds and all the old rules combined.

function getNumDaysFromDates(startDateStr, endDateStr) {
  const start = new Date(startDateStr); // "2025-03-01"
  const end   = new Date(endDateStr);   // "2025-03-10"

  const MS_PER_DAY = 1000 * 60 * 60 * 24;
  const diffMs = end - start;

  // inclusive: 1–1 = 1 day
  return Math.max(1, Math.ceil(diffMs / MS_PER_DAY) + 1);
}

function getDailyServiceHours({ morning, midday, night }) {
  const count = [morning, midday, night].filter(Boolean).length;

  if (night) {
    // Any option that includes dinner is treated as a full day
    return 8;
  } else if (morning && midday) {
    return 5;
  } else if (count === 1) {
    return 3;
  } else {
    return 0;
  }
}

function getChefRateByHours(totalHours) {
  // New rule:
  // > 140 hours → monthly tier (250k)
  // > 24 hours  → weekly tier (350k)
  // else        → normal (800k)
  if (totalHours > 140) {
    return { hourly: 250000, tier: "monthly" };
  } else if (totalHours > 24) {
    return { hourly: 350000, tier: "weekly" };
  } else {
    return { hourly: 800000, tier: "daily" };
  }
}

function calculateBooking(params) {
  let {
    numDays,
    startDate,  // optional "YYYY-MM-DD"
    endDate,    // optional "YYYY-MM-DD"
    guestCount,
    morning,
    midday,
    night,
  } = params;

  // 1) Derive numDays from dates if needed
  if (!numDays && startDate && endDate) {
    numDays = getNumDaysFromDates(startDate, endDate);
  }

  if (!numDays || numDays <= 0) {
    throw new Error("numDays must be at least 1");
  }

  // 2) Service hours per day
  const dailyServiceHours = getDailyServiceHours({ morning, midday, night });

  // 3) Service + shopping hours
  const serviceHours   = dailyServiceHours * numDays;
  const shoppingBlocks = Math.ceil(numDays / 7); // 1 block per started week
  const shoppingHours  = shoppingBlocks * 2;     // 2h per block
  const totalChefHours = serviceHours + shoppingHours;

  // 4) Chef rate based on total hours (new rule)
  const { hourly: chefHourly, tier: pricingTier } = getChefRateByHours(totalChefHours);

  // 5) Helper logic
  const helperHourly = 400000;
  let helperHours = 0;

  if (guestCount > 10) {
    helperHours = totalChefHours; // same hours as chef
  }

  // 6) Costs
  const chefCost   = totalChefHours * chefHourly;
  const helperCost = helperHours     * helperHourly;
  const totalCost  = chefCost + helperCost;

  return {
    // Inputs / basics
    numDays,
    guestCount,

    // Hours breakdown
    dailyServiceHours,
    serviceHours,
    shoppingBlocks,
    shoppingHours,
    totalChefHours,
    helperHours,

    // Pricing tier
    pricingTier,     // "daily" | "weekly" | "monthly"
    chefHourly,
    helperHourly,

    // Money
    chefCost,
    helperCost,
    totalCost,
  };
}
